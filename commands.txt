=========================
CONTAINER MANAGEMENT
=========================

docker ps --> List all the running containers
docker ps -a --> List all the running + stopped containers
docker ps -aq --> List all the running + stopped container IDs
docker start <containerId> --> Start a stopped container
docker stop <containerId> --> Stop a running container
docker restart <containerId> --> Restart a container
docker stop $(docker ps -aq) --> Stop all running containers
docker rm <containerId> --> Remove a stopped container
docker rm -f <containerId> --> Force stop and remove a container
docker exec -it <containerId> /bin/bash --> Open a Bash terminal inside a running container
docker logs <containerId> --> Show logs produced by a container
docker inspect <id> --> Show detailed low-level information about a container or image
docker top <containerId> --> Show running processes inside a container
docker stats --> Display live CPU, memory, network, and disk usage for running containers

=========================
IMAGE MANAGEMENT
=========================

docker images --> List all the locally stored images
docker pull node:latest --> Pull the latest Node.js image from Docker Hub
docker build -t <app-name>:<tag> . --> Build a Docker image from the Dockerfile in the current directory
docker rmi <imageId> --> Remove a locally stored image by ID
docker image prune --> Remove all unused (dangling) images
docker history <image-name> --> Show image layers and build history
docker tag <old-image> <new-image> --> Add a new tag/name to an existing image

=========================
RUNNING CONTAINERS
=========================

docker run -it <app-name>:<tag> /bin/bash --> Run a container interactively with a Bash terminal
docker run -d <app-name>:<tag> --> Run a container in detached (background) mode
docker run -d -p <host-port>:<container-port> <app-name>:<tag> --> Run a container and publish a port
docker run -d -p <host-port>:<container-port> --name <container-name> <app-name>:<tag> --> Run a container, publish a port, and assign a container name
docker run -d -p <host-port>:<container-port> --name <container-name> -v <volume-name>:<mount-path> <app-name>:<tag> --> Run a container with a named volume attached
docker run -d -p <host-port>:<container-port> --network <network-name> <app-name>:<tag> --> Run a container on a specific network

=========================
VOLUME MANAGEMENT
=========================

docker volume ls --> List all Docker volumes
docker volume create <volume-name> --> Create a new named volume
docker volume rm <volume-name> --> Remove a named volume
docker volume inspect <volume-name> --> Show detailed information about a volume
docker volume prune --> Remove all unused volumes

=========================
NETWORK MANAGEMENT
=========================

docker network ls --> List all Docker networks
docker network create <network-name> --> Create a new network
docker network rm <network-name> --> Remove a network
docker network connect <network-name> <container> --> Connect a container to a network
docker network disconnect <network-name> <container> --> Disconnect a container from a network
docker network inspect <network-name> --> Show detailed information about a network
docker network prune --> Remove all unused networks

=========================
DOCKER COMPOSE
=========================

docker compose build --> Build images defined in docker-compose.yml
docker compose up --> Build (if needed) and start all services
docker compose up -d --> Build (if needed) and start all services in detached mode
docker compose down --> Stop and remove all containers, networks, and default Compose resources
docker compose down -v --> Stop and remove containers, networks, and associated volumes
docker compose ps --> List containers managed by the current Compose project
docker compose ls --> List all Compose projects
docker compose logs --> Show logs for all services
docker compose logs <service-name> --> Show logs for a specific service
docker compose restart --> Restart all services
docker compose stop --> Stop all services without removing them
docker compose start --> Start previously stopped services

=========================
DOCKER HUB
=========================

docker login --> Log in to Docker Hub
docker logout --> Log out of Docker Hub
docker push <username>/<image-name>:<tag> --> Push an image to Docker Hub

=========================
SYSTEM CLEANUP
=========================

docker system df --> Show Docker disk usage
docker system prune --> Remove all unused containers, networks, images, and optionally volumes
docker system prune -a --> Remove all unused images (not just dangling), containers, and networks
